const pClass={ok:'p-ok',caution:'p-cau'};

function makeThumb(stop, container) {
  const img = document.createElement('img');
  img.alt = stop.name;
  img.loading = 'lazy';
  img.style.cssText = 'width:100%;height:100%;object-fit:cover;display:block';
  let tried = 0;
  img.onerror = function() {
    tried++;
    if (tried === 1 && stop.fb) { img.src = stop.fb; }
    else { img.remove(); container.textContent = stop.emoji; }
  };
  img.src = stop.img;
  container.appendChild(img);
}

function buildStops() {
  Object.entries(DAYS).forEach(([dayId, day]) => {
    const container = document.getElementById('stops-' + dayId);
    if (!container) return;
    day.stops.forEach((stop, idx) => {
      const card = document.createElement('button');
      card.className = 'card';
      card.onclick = () => openModal(stop);

      const num = document.createElement('div');
      num.className = 'cnum';
      num.style.cssText = `background:${day.numBg};border:1.5px solid ${day.numBorder};color:${day.accent}`;
      num.textContent = idx + 1;

      const thumb = document.createElement('div');
      thumb.className = 'cthumb';
      makeThumb(stop, thumb);

      const info = document.createElement('div');
      info.className = 'cinfo';
      info.innerHTML = `
        <div class="cbadges">
          <span class="ctag" style="background:${stop.tagColor}18;color:${stop.tagColor};border:1px solid ${stop.tagColor}44">${stop.tag}</span>
          <span class="ctime">${stop.time}</span>
          <span class="pbadge ${pClass[stop.preg]}">${pLabel[stop.preg]}</span>
        </div>
        <div class="cname">${stop.name}</div>
        <div class="ctype">${stop.emoji} ${stop.type}</div>`;

      const arrow = document.createElement('div');
      arrow.className = 'carrow';
      arrow.textContent = '›';

      card.appendChild(num);
      card.appendChild(thumb);
      card.appendChild(info);
      card.appendChild(arrow);
      container.appendChild(card);
    });
  });
}

function openModal(stop) {
  document.getElementById('memoji').textContent = stop.emoji;
  const mimg = document.getElementById('mimg');
  mimg.style.display = 'block';
  let tried = 0;
  mimg.onerror = function() {
    tried++;
    if (tried === 1 && stop.fb) { mimg.src = stop.fb; }
    else { mimg.style.display = 'none'; }
  };
  mimg.src = stop.img;

  const tag = document.getElementById('mtag');
  tag.textContent = stop.tag;
  tag.style.cssText = `background:${stop.tagColor}22;color:${stop.tagColor};border:1.5px solid ${stop.tagColor}55;font-size:10px;font-weight:600;padding:3px 10px;border-radius:6px;letter-spacing:.5px`;

  document.getElementById('mtype').textContent = stop.emoji + ' ' + stop.type;
  document.getElementById('mtitle').textContent = stop.name;
  document.getElementById('mdur').textContent = '🕐 ' + stop.duration;
  document.getElementById('mdesc').textContent = stop.desc;

  const pe = document.getElementById('mpreg');
  pe.style.cssText = `background:${pBg[stop.preg]};border:1.5px solid ${pBor[stop.preg]};border-radius:12px;padding:13px 16px;font-size:13px;line-height:1.65;font-weight:300;margin-bottom:16px`;
  pe.innerHTML = `<div style="font-size:10px;letter-spacing:2px;text-transform:uppercase;color:${pCol[stop.preg]};margin-bottom:7px;font-weight:600">🤰 Pregnancy (1–3 months) &nbsp;<span class="pbadge ${pClass[stop.preg]}">${pLabel[stop.preg]}</span></div>${stop.pregNote}`;

  document.getElementById('mtips').textContent = stop.tips;

  // Reset scroll position on modal
  document.getElementById('modal-card').scrollTop = 0;

  document.getElementById('overlay').classList.add('open');
  document.body.style.overflow = 'hidden';
}

function closeModal() {
  document.getElementById('overlay').classList.remove('open');
  document.body.style.overflow = '';
}

function switchDay(dayId) {
  document.querySelectorAll('.panel').forEach(p => p.classList.remove('active'));
  document.getElementById('panel-' + dayId).classList.add('active');
  const order = ['sun','mon','tue','wed','extra'];
  const classMap = {sun:'a-sun',mon:'a-mon',tue:'a-tue',wed:'a-wed',extra:'a-extra'};
  document.querySelectorAll('.tab').forEach((t, i) => {
    t.className = 'tab';
    if (order[i] === dayId) t.classList.add(classMap[dayId]);
  });
}

// Close on Escape key
document.addEventListener('keydown', e => { if (e.key === 'Escape') closeModal(); });
